/**
 * Called once at account_created time.
 * Schedules all 5 drip emails using
 * setTimeout. This works on Replit's
 * always-on server. If the server restarts,
 * the email_campaign_log table prevents
 * re-sending already-sent steps.
 */
export async function startOnboardingCampaign(args: {
  userId: string;
  email: string;
  firstName?: string;
}): Promise<void> {
  for (const step of STEPS) {
    setTimeout(async () => {
      await sendCampaignStep({
        step,
        userId: args.userId,
        email: args.email,
        firstName: args.firstName,
      });
    }, step.delayMs);
  }
  console.log(
    `[onboarding] campaign scheduled` +
    ` for ${args.email}`
  );
}

async function sendCampaignStep(args: {
  step: CampaignStep;
  userId: string;
  email: string;
  firstName?: string;
}): Promise<void> {
  const { step, userId, email, firstName } = args;
  const name = firstName?.trim() ?? "";

  // Check if already sent (server restart
  // protection via Supabase log)
  try {
    const { supabaseAdmin } = await import(
      "../supabase"
    );
    if (!supabaseAdmin) return;

    const { data: existing } =
      await supabaseAdmin
        .from("email_campaign_log")
        .select("id")
        .eq("user_id", userId)
        .eq("campaign_id", CAMPAIGN_ID)
        .eq("step_number", step.stepNumber)
        .maybeSingle();

    if (existing) {
      console.log(
        `[onboarding] step ${step.stepNumber}` +
        ` already sent to ${email} — skip`
      );
      return;
    }

    // Send the email
    const resend = getResend();
    if (!resend) {
      console.warn(
        "[onboarding] RESEND_API_KEY not set"
      );
      return;
    }

    const from =
      process.env.ALERT_FROM_EMAIL ||
      "Havo <updates@havofl.com>";

    const { error } = await resend.emails.send({
      from,
      to: email,
      subject: step.subject(name),
      html: step.html(name),
      text: step.text(name),
      reply_to: "christian@havo.com",
    } as any);

    if (error) {
      console.error(
        `[onboarding] step ${step.stepNumber}` +
        ` send error:`, error
      );
      return;
    }

    // Log the send
    await supabaseAdmin
      .from("email_campaign_log")
      .insert({
        user_id: userId,
        email,
        campaign_id: CAMPAIGN_ID,
        step_number: step.stepNumber,
        status: "sent",
      });

    console.log(
      `[onboarding] step ${step.stepNumber}` +
      ` sent to ${email}`
    );
  } catch (err: any) {
    console.error(
      `[onboarding] step ${step.stepNumber}` +
      ` error:`, err?.message
    );
  }
}

function getResend() {
  const key = process.env.RESEND_API_KEY;
  if (!key) return null;
  const { Resend } = require("resend");
  return new Resend(key);
}
