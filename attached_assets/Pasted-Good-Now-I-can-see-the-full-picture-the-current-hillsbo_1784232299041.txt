/**
 * Hillsborough County property tax estimator.
 *
 * Calls the HCPA CommonServices API —
 * the same backend used by the official
 * Tax Estimator at:
 * gis.hcpafl.org/PropertySearch/TaxEstimator
 *
 * Two-step flow:
 *   1. BasicSearch → PIN number
 *   2. TaxEstimator?pin → millage rates
 *      + nonAdValoremTaxes (CDD/assessments)
 *
 * The low estimate matches HCPA's own formula:
 *   taxableBase = purchasePrice × 0.85
 *   Apply FL homestead exemption
 *   Add nonAdValoremTaxes on top
 */

const HCPA_BASE =
  "https://gis.hcpafl.org/CommonServices/" +
  "property/search";

const HCPA_HEADERS: Record<string, string> = {
  Referer:
    "https://gis.hcpafl.org/PropertySearch/" +
    "TaxEstimator.aspx",
  Accept:
    "application/json, text/plain, */*",
  "X-Requested-With": "XMLHttpRequest",
};

const TIMEOUT_MS = 8_000;

/** Extract street portion for BasicSearch.
 *  "3102 W Nassau St, Tampa, FL 33607"
 *  → "3102 W Nassau"
 *  (3 tokens avoids suffix mismatches) */
function streetForSearch(
  fullAddress: string
): string {
  const street = fullAddress
    .split(",")[0]
    .trim();
  return street
    .split(/\s+/)
    .slice(0, 3)
    .join(" ");
}

// ── Tax calculation (mirrors HCPA JS) ────

/**
 * Calculates ad valorem tax using HCPA's
 * exact formula.
 *
 * HCPA low estimate = purchasePrice × 0.85
 * as the taxable base. Homestead exemption:
 *   - $25k off school + non-school taxes
 *   - $25k more off non-school only
 *     (value $50k–$75k range)
 */
export function calcAdValorem(
  purchasePrice: number,
  homestead: boolean,
  schoolRate: number,
  nonSchoolRate: number,
  totalRate: number
): number {
  const taxable = purchasePrice * 0.85;

  if (!homestead) {
    return Math.round(
      taxable * (totalRate / 1000)
    );
  }

  const schoolTaxable = Math.max(
    0, taxable - 25_000
  );
  const schoolTax =
    schoolTaxable * (schoolRate / 1000);

  const nonSchoolTaxable =
    taxable < 50_000
      ? Math.max(0, taxable - 25_000)
      : Math.max(0, taxable - 50_000);
  const nonSchoolTax =
    nonSchoolTaxable * (nonSchoolRate / 1000);

  return Math.round(schoolTax + nonSchoolTax);
}

// ── API calls ─────────────────────────────

/** Step 1: address → PIN via BasicSearch. */
async function lookupPIN(
  address: string
): Promise<string | null> {
  const query = encodeURIComponent(
    streetForSearch(address)
  );
  const url =
    `${HCPA_BASE}/BasicSearch?address=` +
    query;

  const resp = await fetch(url, {
    headers: HCPA_HEADERS,
    signal: AbortSignal.timeout(TIMEOUT_MS),
  });
  if (!resp.ok) return null;

  const results = await resp.json();
  if (
    !Array.isArray(results) ||
    !results.length
  ) return null;

  const inputCity = (
    address.split(",")[1] ?? ""
  ).trim().toUpperCase();

  const best =
    results.find(
      (r: any) =>
        (r.address ?? "")
          .toUpperCase()
          .includes(inputCity)
    ) ?? results[0];

  return best?.pin ?? null;
}

/** Step 2: PIN → tax data. */
async function fetchTaxData(
  pin: string
): Promise<HCPAApiTaxData | null> {
  const url =
    `${HCPA_BASE}/TaxEstimator?pin=` +
    encodeURIComponent(pin);

  const resp = await fetch(url, {
    headers: HCPA_HEADERS,
    signal: AbortSignal.timeout(TIMEOUT_MS),
  });
  if (!resp.ok) return null;

  const data = await resp.json();

  if (
    typeof data?.schoolTaxRate !== "number" ||
    typeof data?.nonschoolTaxRate !== "number"
  ) return null;

  return {
    schoolTaxRate: data.schoolTaxRate,
    nonschoolTaxRate: data.nonschoolTaxRate,
    totalTaxRate: data.totalTaxRate,
    nonAdValoremTaxes:
      data.nonAdValoremTaxes ?? 0,
    taxDistrict: data.taxDistrict ?? "",
    justValue: data.justValue ?? 0,
    assessedValue: data.assessedValue ?? 0,
  };
}

/**
 * Derives the ad valorem percentage of
 * purchase price for storage in Supabase.
 *   adValoremPct = adValoremTax / purchasePrice
 */
export function deriveTaxPct(
  adValoremTax: number,
  purchasePrice: number
): number {
  if (purchasePrice <= 0) return 0;
  return adValoremTax / purchasePrice;
}

/**
 * Recalculates the full tax estimate from
 * cached millage rates for any new price.
 * Uses the accurate millage formula so
 * homestead exemptions are correct at
 * any price. Falls back to pct × price
 * if rates unavailable.
 */
export function recalcFromCache(params: {
  purchasePrice: number;
  isPrimaryResidence: boolean;
  schoolTaxRate: number;
  nonSchoolTaxRate: number;
  totalMillageRate: number;
  adValoremPct: number;
  nonAdValoremAmt: number;
}): {
  annualTax: number;
  monthlyTax: number;
  adValoremTax: number;
  nonAdValoremTax: number;
  source: "cache-millage" | "cache-pct";
} {
  const {
    purchasePrice,
    isPrimaryResidence,
    schoolTaxRate,
    nonSchoolTaxRate,
    totalMillageRate,
    adValoremPct,
    nonAdValoremAmt,
  } = params;

  let adValoremTax: number;
  let source: "cache-millage" | "cache-pct";

  if (schoolTaxRate > 0 &&
      nonSchoolTaxRate > 0) {
    adValoremTax = calcAdValorem(
      purchasePrice,
      isPrimaryResidence,
      schoolTaxRate,
      nonSchoolTaxRate,
      totalMillageRate
    );
    source = "cache-millage";
  } else {
    adValoremTax = Math.round(
      purchasePrice * adValoremPct
    );
    source = "cache-pct";
  }

  const annualTax =
    adValoremTax + nonAdValoremAmt;

  return {
    annualTax,
    monthlyTax:
      Math.round((annualTax / 12) * 100) / 100,
    adValoremTax,
    nonAdValoremTax: nonAdValoremAmt,
    source,
  };
}
