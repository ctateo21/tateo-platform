Add helper:

function isRichEnough(analysis) {
  return (
    analysis?.status_label !== "Insufficient Data" &&
    analysis?.market_comps?.comps?.length > 0 &&
    analysis?.projected_sale_price?.projected_low &&
    analysis?.projected_sale_price?.projected_high &&
    analysis?.data_sources?.some(source =>
      String(source.name || "").toLowerCase().includes("anthropic web search") &&
      source.status === "connected"
    )
  );
}
