1. Adds a 30-day policy effective date to all
   QuoteRUSH quote requests
2. Auto-triggers quotes the moment any address
   is entered on any page (no button click needed)
3. Stores quote results in a shared Postgres DB
   table so two users searching the same address
   share the same quote (no duplicate API cost)
4. Enforces a 30-day cooldown on re-running
   quotes for the same address
5. Adds a Refresh button that fetches the latest
   carrier results from QuoteRUSH without
   triggering a new submission (free, instant)
6. Fixes the "only 1 carrier showing" issue by
   continuing to poll until results stabilize
7. Keeps top 3 carrier quotes visible for 30 days
   then prompts re-run when expired

// Shared QuoteRUSH auto-trigger utility.
// Uses localStorage as a fast local cache
// backed by the server DB cache.
// Safe to call from any page — deduplicates
// automatically.

const LS_PREFIX = "qr_auto_";
const LS_TTL_MS = 30 * 24 * 60 * 60 * 1000; // 30 days

export interface QRCacheEntry {
  address: string;
  leadId: number | null;
  status: "pending" | "success" | "error";
  quotes: Array<{
    siteName: string;
    annualPremium: number;
    monthlyPremium: number;
    coverageA: number;
    hurricaneDeductible: string;
    aop: string;
    quoteUrl: string | null;
    quoteDate: string;
    rank: number;
  }>;
  quoteCounter: number;
  coverageA: number;
  expiresAt: string; // ISO
  triggeredAt: string; // ISO
}

function lsKey(address: string): string {
  return (
    LS_PREFIX +
    address
      .trim()
      .toLowerCase()
      .replace(/\s+/g, "_")
      .slice(0, 80)
  );
}

export function getQRCache(
  address: string
): QRCacheEntry | null {
  if (!address) return null;
  try {
    const raw = localStorage.getItem(
      lsKey(address)
    );
    if (!raw) return null;
    const e = JSON.parse(raw) as QRCacheEntry;
    if (new Date(e.expiresAt) < new Date()) {
      localStorage.removeItem(lsKey(address));
      return null;
    }
    return e;
  } catch {
    return null;
  }
}

export function setQRCache(
  address: string,
  patch: Partial<QRCacheEntry>
): void {
  if (!address) return;
  try {
    const key = lsKey(address);
    const existing = getQRCache(address);
    const expiresAt =
      patch.expiresAt ??
      existing?.expiresAt ??
      (() => {
        const d = new Date();
        d.setDate(d.getDate() + 30);
        return d.toISOString();
      })();
    const next: QRCacheEntry = {
      address,
      leadId: null,
      status: "pending",
      quotes: [],
      quoteCounter: 0,
      coverageA: 0,
      triggeredAt: new Date().toISOString(),
      ...(existing ?? {}),
      ...patch,
      expiresAt,
    };
    localStorage.setItem(
      key, JSON.stringify(next)
    );
  } catch {}
}

export function isQRExpired(
  address: string
): boolean {
  const e = getQRCache(address);
  if (!e) return true;
  return new Date(e.expiresAt) < new Date();
}

export function clearQRCache(
  address: string
): void {
  try {
    localStorage.removeItem(lsKey(address));
  } catch {}
}

/**
 * Checks the server DB cache for an address.
 * If found and valid, hydrates localStorage.
 * Returns the cache entry or null.
 */
export async function checkServerCache(
  address: string
): Promise<QRCacheEntry | null> {
  if (!address) return null;
  try {
    const res = await fetch(
      `/api/insurance/qr-cache?address=` +
        encodeURIComponent(address)
    );
    const data = await res.json();
    if (!data.found || data.expired) return null;

    const entry: QRCacheEntry = {
      address,
      leadId: data.leadId ?? null,
      status: data.status ?? "pending",
      quotes: data.quotes ?? [],
      quoteCounter: data.quoteCounter ?? 0,
      coverageA: data.coverageA ?? 0,
      expiresAt:
        data.expiresAt ??
        (() => {
          const d = new Date();
          d.setDate(d.getDate() + 30);
          return d.toISOString();
        })(),
      triggeredAt:
        data.triggeredAt ??
        new Date().toISOString(),
    };

    // Hydrate localStorage from server
    setQRCache(address, entry);
    return entry;
  } catch {
    return null;
  }
}

/**
 * Main trigger function. Call from any page
 * when address + price are confirmed.
 * Safe to call multiple times — deduplicates.
 */
export async function triggerAutoQuote(opts: {
  address: string;
  price?: number;      // full property value
  coverageA?: number;  // or Coverage A directly
  policyType?: string;
  floodZone?: string;
  isAuthenticated: boolean;
}): Promise<void> {
  const {
    address,
    isAuthenticated,
  } = opts;

  if (!isAuthenticated) return;
  if (!address || address === "Unknown Address")
    return;

  const coverageA =
    opts.coverageA ??
    Math.round((opts.price ?? 0) * 0.85);

  if (coverageA <= 0) return;

  // Check localStorage first (fast)
  const cached = getQRCache(address);
  if (
    cached &&
    (cached.status === "pending" ||
      cached.status === "success")
  ) return;

  // Check server DB cache
  const serverCached =
    await checkServerCache(address);
  if (
    serverCached &&
    (serverCached.status === "pending" ||
      serverCached.status === "success")
  ) return;

  // Not cached anywhere — write pending and trigger
  setQRCache(address, {
    status: "pending",
    coverageA,
    leadId: null,
    quotes: [],
    quoteCounter: 0,
    triggeredAt: new Date().toISOString(),
  });

  console.log(
    "[qr-auto] triggering:", address,
    "coverageA:", coverageA
  );

  try {
    const res = await fetch(
      "/api/insurance/qr-start",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          address,
          coverageA,
          policyType: opts.policyType || "HO3",
          yearIdx: 1,
          roofIdx: 1,
          constIdx: 0,
          windIdx: 1,
          hurrIdx: 0,
          claimsIdx: 0,
          aopDeductible: 2500,
          floodZone: opts.floodZone || "X",
          sqFt: 0,
          newPurchase: false,
        }),
      }
    );

        ...(data.fromCache &&
          data.quotes?.length > 0
          ? {
              status: "success" as const,
              quotes: data.quotes,
              quoteCounter: data.quoteCounter,
              expiresAt: data.expiresAt,
            }
          : {}),
      });
    } else {
      setQRCache(address, { status: "error" });
    }
  } catch (e) {
    console.error("[qr-auto] error:", e);
    setQRCache(address, { status: "error" });
  }
}

  function startPolling(leadId: number): void {
    if (qrPollRef.current)
      clearInterval(qrPollRef.current);
    if (qrTimerRef.current)

    setQrElapsed(0);
    qrPrevCounterRef.current = 0;
    qrStableRef.current = 0;

    qrTimerRef.current = setInterval(() => {
      setQrElapsed((p) => p + 1);
    }, 1000);

    let pollCount = 0;
    const MAX_POLLS = 24; // 12 minutes

    const doPoll = async () => {
      pollCount++;
      try {
        const res = await fetch(
          "/api/insurance/qr-quotes",
          {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
            },
            body: JSON.stringify({
              leadId,
              address,
            }),
          }
        );
        const data = await res.json();
        const counter: number =
          data.quoteCounter ?? 0;

        if (counter > 0) {
          setQrQuotes(data.quotes ?? []);
          setQrQuoteCounter(counter);
          setQrStatus("success");
          setQRCache(address, {
            status: "success",
            quotes: data.quotes ?? [],
            quoteCounter: counter,
          });

          // Stable detection — stop only after
          // 3 consecutive polls with same count
          // (gives carriers time to finish)
          if (
            counter === qrPrevCounterRef.current
          ) {
            qrStableRef.current += 1;
          } else {
            qrStableRef.current = 0;
          }
          qrPrevCounterRef.current = counter;

          if (
            qrStableRef.current >= 3 ||
            pollCount >= MAX_POLLS
          ) {
            if (qrPollRef.current)
              clearInterval(qrPollRef.current);
            if (qrTimerRef.current)
