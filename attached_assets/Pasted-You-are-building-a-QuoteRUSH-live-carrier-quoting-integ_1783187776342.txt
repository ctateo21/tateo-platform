function parseLeadId(responseText: string): number | null {
  // Try JSON first
  try {
    const json = JSON.parse(responseText);
    const id = json.id || json.leadId || json.LeadId 
              || json.lead_id || json.Id;
    if (id) return parseInt(String(id), 10);
  } catch {}

  // Try to extract any integer from the text
  const match = responseText.match(/\d+/);
  if (match) return parseInt(match[0], 10);
  return null;
}

export async function importAndSubmit(
  params: QuoteRushParams
): Promise<{ leadId: number; submitted: boolean; 
             error?: string }> {
  const WEBID = process.env.QUOTERUSH_WEBID!;
  const WEBPASSWORD = process.env.QUOTERUSH_WEBPASSWORD!;
  const ENDPOINT_KEY = process.env.QUOTERUSH_ENDPOINT_KEY!;
  const AGENCY = process.env.QUOTERUSH_AGENCY!;
  const ASSIGNED_EMAIL = 
    process.env.QUOTERUSH_ASSIGNED_EMAIL 
    || "christian@tateoco.com";

  if (!WEBID || !WEBPASSWORD || !ENDPOINT_KEY || !AGENCY) {
    return { 
      leadId: 0, submitted: false,
      error: "QuoteRUSH env vars not configured" 
    };
  }

  // STEP 1: Import lead
  console.log("[quoterush] importing lead for:", 
    params.address);
  const importPayload = buildImporterPayload(
    params, ASSIGNED_EMAIL
  );
  const importRes = await fetch(
    `https://importer.quoterush.com/Json/Import/${WEBID}`,
    {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "webpassword": WEBPASSWORD,
      },
      body: JSON.stringify(importPayload),
    }
  );
  const importText = await importRes.text();
  console.log("[quoterush] import response:", importText);

  if (!importRes.ok) {
    return {
      leadId: 0, submitted: false,
      error: `Import failed ${importRes.status}: ${importText}`
    };
  }

  // Extract LeadId from response
  let leadId = parseLeadId(importText);

  // Fallback: use GetNewLeads to find the lead we just created
  if (!leadId || leadId <= 0) {
    console.log("[quoterush] extracting LeadId via GetNewLeads");
    try {
      const newLeadsRes = await fetch(
        "https://api.quoterush.com/GetNewLeads",
        {
          method: "GET",
          headers: {
            "WebId": WEBID,
            "WebIdPassword": WEBPASSWORD,
          },
        }
      );
      const newLeadsText = await newLeadsRes.text();
      console.log("[quoterush] GetNewLeads:", newLeadsText);
      const newLeadsJson = JSON.parse(newLeadsText);
      const leads = Array.isArray(newLeadsJson) 
        ? newLeadsJson 
        : (newLeadsJson.Leads || newLeadsJson.leads || []);
      if (leads.length > 0) {
        const first = leads[0];
        leadId = parseInt(
          String(first.id || first.Id || first.leadId 
                || first.LeadId || 0), 
          10
        );
      }
    } catch (e) {
      console.error("[quoterush] GetNewLeads error:", e);
    }
  }

  if (!leadId || leadId <= 0) {
    return { 
      leadId: 0, submitted: false,
      error: "Could not extract LeadId from response" 
    };
  }

  console.log("[quoterush] LeadId:", leadId);

  // STEP 2: Submit quote request (triggers QuoteBot)
  const submitBody = {
    EndpointKey: ENDPOINT_KEY,
    Agency: AGENCY,
    LeadId: leadId,
    LOB: "Home",
    Submitter: ASSIGNED_EMAIL,
  };
  const submitRes = await fetch(
    "https://api.quoterush.com/SubmitQuoteRequest",
    {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(submitBody),
    }
  );
  const submitText = await submitRes.text();
  console.log("[quoterush] submit response:", submitText);

  return {
    leadId,
    submitted: submitRes.ok,
    error: submitRes.ok ? undefined : submitText,
  };
}

export async function getQuotes(
  leadId: number
): Promise<QuoteRushResult> {
  const ENDPOINT_KEY = process.env.QUOTERUSH_ENDPOINT_KEY!;
  const AGENCY = process.env.QUOTERUSH_AGENCY!;

  try {
    const res = await fetch(
      "https://api.quoterush.com/GetQuotes",
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          EndpointKey: ENDPOINT_KEY,
          Agency: AGENCY,
          LeadId: leadId,
          LOB: "Home",
        }),
      }
    );
    const text = await res.text();
    console.log("[quoterush] GetQuotes response:", text);

    if (!res.ok) {
      return {
        status: "error", quotes: [], leadId, 
        quoteCounter: 0,
        errorMessage: `GetQuotes failed: ${text}`,
      };
    }

    const json = JSON.parse(text);
    const rawQuotes = json.Quotes || [];
    const quoteCounter = json.QuoteCounter || 0;

    if (quoteCounter === 0) {
      return { 
        status: "pending", quotes: [], 
        leadId, quoteCounter: 0 
      };
    }

    // Parse and sort by premium ascending
    const quotes: QuoteRushQuote[] = rawQuotes
      .filter((q: any) => q.Premium && 
        parseFloat(q.Premium) > 0)
      .map((q: any) => {
        const annual = parseFloat(q.Premium || "0");
        return {
          siteName: q.SiteName || "Unknown",
          premium: annual,
          annualPremium: Math.round(annual),
          monthlyPremium: Math.round(annual / 12),
          coverageA: parseInt(q.CoverageA || "0", 10),
          hurricaneDeductible: q.HurricaneDeductible || "",
          aop: q.AOP || "",
          quoteUrl: q.QuoteURL || null,
          quoteMessage: q.QuoteMessage || "",
          quoteDate: q.QuoteDate || "",
          rank: 0,
        };
      })
      .sort((a: QuoteRushQuote, b: QuoteRushQuote) => 
        a.annualPremium - b.annualPremium
      )
      .map((q: QuoteRushQuote, i: number) => ({
        ...q, rank: i + 1
      }));

    return {
      status: quotes.length > 0 ? "success" : "pending",
      quotes,
      leadId,
      quoteCounter,
    };

  } catch (err: any) {
    console.error("[quoterush] GetQuotes error:", err);
    return {
      status: "error", quotes: [], leadId, quoteCounter: 0,
      errorMessage: err?.message,
    };
  }
}

function parseQRAddress(fullAddress: string) {
  const parts = fullAddress.split(",").map(p => p.trim());
  const streetAddress = parts[0] || "";
  const city = parts[1] || "";
  const last = parts[parts.length - 1] || "";
  const m = last.match(/([A-Z]{2})\s+(\d{5})/);
  return {
    streetAddress,
    city,
    state: m?.[1] || "FL",
    zip: m?.[2] || "",
    county: "",
  };
}

  async function startQuoteRush() {
    if (!address || !rebuild || !isAuthenticated) return;

    // Clear any existing polling
    if (qrPollRef.current) clearInterval(qrPollRef.current);
    if (qrTimerRef.current) clearInterval(qrTimerRef.current);

    setQrStatus("starting");
    setQrQuotes([]);
    setQrQuoteCounter(0);
    setQrElapsed(0);

    // Start elapsed timer
    qrTimerRef.current = setInterval(() => {
      setQrElapsed(prev => prev + 1);
    }, 1000);

    try {
      // Step 1: Import lead + submit to QuoteBot
      const startRes = await fetch(
        "/api/insurance/qr-start",
        {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            address,
            coverageA: rebuild,
            policyType: policyType || "HO3",
            yearIdx,
            roofIdx,
            constIdx,
            windIdx,
            hurrIdx,
            claimsIdx,
            aopDeductible,
            floodZone: floodZone || "X",
            sqFt: 0,
            newPurchase: false,
          }),
        }
      );

      const startData = await startRes.json();

      if (!startRes.ok || !startData.leadId) {
        setQrStatus("error");
        if (qrTimerRef.current) 
          clearInterval(qrTimerRef.current);
        return;
      }

      const leadId = startData.leadId;
      setQrLeadId(leadId);
      setQrStatus("pending");

      // Step 2: Poll GetQuotes every 30 seconds
      // QuoteBot processes carriers one at a time
      // so quotes arrive progressively over 2-10 mins
      let pollCount = 0;
      const MAX_POLLS = 20; // 10 minutes max

      const poll = async () => {
        pollCount++;
        try {
          const quotesRes = await fetch(
            "/api/insurance/qr-quotes",
            {
              method: "POST",
              headers: { 
                "Content-Type": "application/json" 
              },
              body: JSON.stringify({ leadId }),
            }
          );
          const quotesData = await quotesRes.json();
          const counter = quotesData.quoteCounter || 0;

          if (counter > 0) {
            setQrQuotes(quotesData.quotes || []);
            setQrQuoteCounter(counter);
            setQrStatus("success");
          }

          // Stop polling if max reached or we have quotes
          // and no more are expected (QuoteCounter stable)
          if (pollCount >= MAX_POLLS) {
            if (qrPollRef.current) 
              clearInterval(qrPollRef.current);
            if (qrTimerRef.current) 
              clearInterval(qrTimerRef.current);
            if (counter === 0) setQrStatus("error");
          }
        } catch (e) {
          console.error("[qr-poll] error:", e);
        }
      };

      // First poll after 30 seconds
      qrPollRef.current = setInterval(poll, 30000);

    } catch (err) {
      console.error("[qr-start] error:", err);
      setQrStatus("error");
      if (qrTimerRef.current) 
        clearInterval(qrTimerRef.current);
    }
  }
